How to use this guide — you do NOT have to start at Module 1.1
This guide is built to be non-linear. Each module is self-contained with its own prerequisites, skip-check, and outcomes. Use the Placement Quiz below to find your entry point, or jump straight to any module using the Module Index. Every exercise and capstone uses the same realistic BookCart online-bookstore dataset (books.csv, orders.csv, customers.csv) — so you build intuition on one domain instead of context-switching. Generating these CSVs is Exercise 1.1 itself; see the BookCart spec in the shared datasets folder for the exact column schema to match.

Quick Start: Three Ways to Use This Guide

  1. Complete beginner → Start at Module 1.1 and go sequentially.
  2. Some experience → Take the Placement Quiz (next section) to find your entry point.
  3. Targeting a specific skill → Jump via the Module Index or follow a curated Learning Track.

Conditionals & Loops

def stock_status(qty: int) -> str:
    if qty == 0:
        return "out of stock"
    elif qty < 5:
        return "low"
    else:
        return "ok"

for qty in [0, 3, 12]:
    print(qty, stock_status(qty))

# while + break/continue
n, total = 0, 0

    n += 1
    if n % 2 == 0:
        continue
    total += n

Functions: Arguments & Returns

def line_total(price: float, qty: int = 1, *, tax: float = 0.0) -> float:
    """Return the line total including optional tax. tax is keyword-only."""
    subtotal = price * qty
    return round(subtotal * (1 + tax), 2)

print(line_total(12.99, 3))            # 38.97
print(line_total(12.99, 3, tax=0.12))  # 43.65

# *args / **kwargs
def summarize(*prices: float, **labels: str) -> dict:
    return {"count": len(prices), "sum": sum(prices), **labels}

Scope & Pure Functions

def apply_discount(prices: list[float], rate: float) -> list[float]:
    # returns a NEW list; does not mutate the input (pure)
    return [round(p * (1 - rate), 2) for p in prices]

original = [10.0, 20.0]
discounted = apply_discount(original, 0.1)
print(original, discounted)   # [10.0, 20.0] [9.0, 18.0]

Exceptions by Type

def load_price(row: dict) -> float:
    try:
        return float(row["price"])
    except KeyError:
        raise ValueError("row is missing a 'price' field") from None
    except ValueError:
        raise ValueError(f"price is not numeric: {row.get('price')!r}")

# try/except/else/finally
try:
    p = load_price({"price": "12.99"})
except ValueError as e:
    print("bad row:", e)
else:
    print("ok:", p)
finally:
    print("done")

Logging Instead of print

Custom Exceptions

class OutOfStockError(Exception):
    """Raised when an order exceeds available stock."""

def reserve(stock: int, qty: int) -> int:
    if qty > stock:
        raise OutOfStockError(f"requested {qty}, only {stock} available")
    return stock - qty

Classes & Methods

class Book:
    def __init__(self, book_id: int, title: str, price: float, stock: int):
        self.book_id = book_id
        self.title = title
        self.price = price
        self.stock = stock

    def sell(self, qty: int) -> float:
        if qty > self.stock:
            raise ValueError("not enough stock")
        self.stock -= qty
        return round(self.price * qty, 2)

    def __repr__(self) -> str:
        return f"Book({self.title!r}, stock={self.stock})"

Dataclasses & Properties

from dataclasses import dataclass, field

@dataclass
class Order:
    order_id: int
    book: Book
    quantity: int
    discount: float = 0.0

    @property
    def total(self) -> float:
        gross = self.book.price * self.quantity
        return round(gross * (1 - self.discount), 2)

o = Order(1, Book(417, "Dune", 12.99, 10), quantity=3, discount=0.1)
print(o.total)   # 35.07

Generators (lazy iteration)

def read_rows(path: str):
    """Yield rows one at a time — memory-friendly for big files."""
    import csv
    with open(path, newline="", encoding="utf-8") as f:
        for row in csv.DictReader(f):
            yield row

# generator expression: sum revenue without building a list
total = sum(float(r["price"]) * int(r["qty"])
            for r in read_rows("data/orders.csv"))

Decorators

import functools, time

def timed(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper

@timed
def load_all(path: str) -> int:
    return sum(1 for _ in read_rows(path))

from typing import Optional

def find_book(books: list[dict], book_id: int) -> Optional[dict]:
    for b in books:
        if b["id"] == book_id:
            return b
    return None

def revenue_by_cat(rows: list[dict]) -> dict[str, float]:
    out: dict[str, float] = {}
    for r in rows:
        out[r["category"]] = out.get(r["category"], 0.0) + r["revenue"]
    return out

    stock: int

def restock(book: BookRow, amount: int) -> BookRow:
    book["stock"] += amount
    return book

App, Models & Endpoints

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="BookCart API")

class Book(BaseModel):
    book_id: int
    title: str
    price: float
    stock: int

CATALOG: dict[int, Book] = {}

@app.get("/books")
def list_books() -> list[Book]:
    return list(CATALOG.values())

@app.get("/books/{book_id}")
def get_book(book_id: int) -> Book:
    if book_id not in CATALOG:
        raise HTTPException(status_code=404, detail="book not found")
    return CATALOG[book_id]

POST with Validation

class OrderIn(BaseModel):
    book_id: int
    quantity: int

@app.post("/orders", status_code=201)
def place_order(order: OrderIn) -> dict:
    book = CATALOG.get(order.book_id)
    if book is None:
        raise HTTPException(404, "book not found")
    if order.quantity > book.stock:
        raise HTTPException(409, "insufficient stock")
    book.stock -= order.quantity
    return {"ok": True, "remaining": book.stock}

SQLAlchemy ORM

from sqlalchemy import create_engine, ForeignKey, func, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship, Session

class Base(DeclarativeBase):
    pass

class Book(Base):
    __tablename__ = "books"
    book_id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str]
    price: Mapped[float]
    orders: Mapped[list["Order"]] = relationship(back_populates="book")

class Order(Base):
    __tablename__ = "orders"
    order_id: Mapped[int] = mapped_column(primary_key=True)
    book_id: Mapped[int] = mapped_column(ForeignKey("books.book_id"))
    quantity: Mapped[int]
    book: Mapped["Book"] = relationship(back_populates="orders")

engine = create_engine("sqlite:///bookcart.db")
Base.metadata.create_all(engine)

with Session(engine) as s:
    stmt = (select(Book.title, func.sum(Order.quantity))
            .join(Order).group_by(Book.title))
    for title, qty in s.execute(stmt):
        print(title, qty)

Tests, Fixtures & Parametrization

# test_pricing.py
import pytest
from bookcart.pricing import line_total

@pytest.fixture
def sample_book():
    return {"price": 12.99, "stock": 10}

@pytest.mark.parametrize("qty,tax,expected", [
    (1, 0.0, 12.99),
    (3, 0.0, 38.97),
    (2, 0.12, 29.10),
])
def test_line_total(qty, tax, expected):
    assert line_total(12.99, qty, tax=tax) == pytest.approx(expected)

def test_oversell_raises(sample_book):
    with pytest.raises(ValueError):
        line_total(sample_book["price"], -1)
